function openDetail(pid){
  var p = DATA.find(function(d){return d.id===pid;});
  if(!p) return;
  var ms = (p.milestones||[]).map(function(m){
    return '<li>' + (m.done?'✅':'⬜') + escHtml(m.name) + ' <span style="color:#9ca3af;font-size:12px">('+escHtml(m.due)+')</span></li>';
  }).join('');
  var risks = (p.risks||[]).map(function(r){return '<li>'+escHtml(r)+'</li>';}).join('');
  var reuse = (p.reuse||[]).map(function(r){return '<span>'+escHtml(r)+'</span>';}).join('');
  var stack = (p.stack||[]).map(function(s){return '<span>'+escHtml(s)+'</span>';}).join('');

  var html =
    '<div class="modal">'+
      '<div class="mh">'+
        '<div class="nm">'+escHtml(p.name)+'</div>'+
        '<span class="badge">'+escHtml(p.priority||'-')+'</span>'+
        '<button class="close" onclick="closeDetail()">×</button>'+
      '</div>'+
      '<div class="mb">'+
        '<div class="sec">'+
          '<h3>简介</h3>'+
          '<p>'+escHtml(p.desc||'无描述')+'</p>'+
        '</div>'+
        '<div class="sec">'+
          '<h3>基本信息</h3>'+
          '<div class="meta-row"><b>ID</b><span>'+escHtml(p.id)+'</span></div>'+
          '<div class="meta-row"><b>状态</b><span>'+escHtml(p.status||'-')+'</span></div>'+
          '<div class="meta-row"><b>截止</b><span>'+escHtml((p.deadline||'-')+(p.deadline_note?' · '+p.deadline_note:''))+'</span></div>'+
          '<div class="meta-row"><b>下一步</b><span>'+escHtml(p.next||'-')+'</span></div>'+
        '</div>'+
        '<div class="sec">'+
          '<h3>里程碑 ('+(p.milestones? p.milestones.length:0)+' 项)</h3>'+
          '<ul>'+ms+'</ul>'+
        '</div>'+
        (p.risks? '<div class="sec"><h3>风险</h3><ul>'+risks+'</ul></div>' : '')+
        (p.reuse? '<div class="sec"><h3>可复用资产</h3><div class="ms">'+reuse+'</div></div>' : '')+
        (p.stack? '<div class="sec"><h3>技术栈</h3><div class="ms">'+stack+'</div></div>' : '')+
      '</div>'+
    '</div>';

  document.getElementById('detailLayer').innerHTML = html;
  document.getElementById('overlay').classList.add('show');
}

function closeDetail(){
  document.getElementById('overlay').classList.remove('show');
}

function escHtml(s){
  return String(s).replace(/&/g,'&amp;').replace(/</g,'&lt;').replace(/>/g,'&gt;');
}

document.getElementById('overlay').addEventListener('click',function(e){
  if(e.target.id==='overlay') closeDetail();
});
document.addEventListener('keydown',function(e){
  if(e.key==='Escape') closeDetail();
});
